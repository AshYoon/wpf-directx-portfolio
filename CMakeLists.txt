cmake_minimum_required(VERSION 3.21)
project(SceneEngine VERSION 0.2.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "SceneEngine requires Windows x64.")
endif()
option(SCENE_BUILD_SAMPLE "Build the standalone Win32 viewport" ON)
add_subdirectory(native/SceneRenderer)
if(SCENE_BUILD_SAMPLE)
    add_subdirectory(samples/NativeHost)
endif()
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
